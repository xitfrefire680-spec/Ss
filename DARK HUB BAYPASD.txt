local function fmtValue(v)

if v >= 1e9 then return string.format("%.2fB", v/1e9) end
if v >= 1e6 then return string.format("%.2fM", v/1e6) end
if v >= 1e3 then return string.format("%.2fK", v/1e3) end

local function GroundMove(targetXZ, speed, easeOut)
local _, hrp = getChar() if not hrp or not targetXZ then return false end
local groundY = hrp.Position.Y
local goal = Vector3.new(targetXZ.X, groundY, targetXZ.Z)
local start = hrp.Position
local dx = goal.X - start.X local dz = goal.Z - start.Z
local dist = math.sqrt(dx*dx + dz*dz)
if dist < 1 then hrp.CFrame = CFrame.new(goal) hrp.AssemblyLinearVelocity = Vector3.zero hrp.AssemblyAngularVelocity = Vector3.zero return true end
speed = math.clamp(tonumber(speed) or 750, 50, 40000)
local t0 = os.clock() local maxTime = (dist / math.max(speed, 50)) * 2 + 4
local cx, cz = hrp.Position.X, hrp.Position.Z
while not HUB.dead do
local dt = RunService.Heartbeat:Wait()
local tx = goal.X - cx local tz = goal.Z - cz
local rem = math.sqrt(tx*tx + tz*tz) if rem < 1 then break end
local sp2 = speed
if easeOut then local prog = 1 - math.clamp(rem / dist, 0, 1) sp2 = math.max(speed * (1 - prog * 0.8), 35) end
local step = math.min(sp2 * dt, rem)
local ux, uz = tx / rem, tz / rem
cx = cx + ux * step cz = cz + uz * step
local np = Vector3.new(cx, groundY, cz) local lookAt = Vector3.new(cx + ux, groundY, cz + uz)
hrp.CFrame = CFrame.lookAt(np, lookAt)
local realVel = step / math.max(dt, 0.001)
hrp.AssemblyLinearVelocity = Vector3.new(ux * realVel, 0, uz * realVel)
hrp.AssemblyAngularVelocity = Vector3.zero
SetSpoofWalkSpeed(HUB.flySpeed)
if BypassState.integrityState then SpoofState(BypassState.integrityState, HUB.flySpeed * HEADROOM) end
SyncHoverPad()
if os.clock() - t0 > maxTime then break end
end
hrp.CFrame = CFrame.new(goal) hrp.AssemblyLinearVelocity = Vector3.zero hrp.AssemblyAngularVelocity = Vector3.zero
return true
end

GroundMove(Vector3.new(sp0.X, 0, ROAD_Z), spd, false)
GroundMove(Vector3.new(SAFE_X, 0, ROAD_Z), spd, false)

else

end

local function runCycle()
if HUB.busy or HUB.dead or not HUB.running then return end
HUB.busy = true
if isCarryingEgg() then returnHome() HUB.busy = false return end
if HUB.activeTarget and HUB.activeTarget.uid then
local rec = findRecord(HUB.activeTarget.uid)
if rec then
if stealEgg(rec) then HUB.activeTarget = nil returnHome() end
HUB.busy = false return
else HUB.activeTarget = nil end
end
local eggs = getFieldEggs()
if #eggs == 0 then HUB.busy = false return end
local target = nil
if HUB.selectedUid then for _, e in ipairs(eggs) do if e.uid == HUB.selectedUid then target = e break end end end
if not target then
table.sort(eggs, function(a, b) if a.value ~= b.value then return a.value > b.value end return a.rank > b.rank end)
target = eggs[1]
end
HUB.activeTarget = { uid = target.uid, name = target.name }
if stealEgg(target.record) then HUB.activeTarget = nil returnHome() end
HUB.busy = false
end

local CHICKEN_EGG_PATH = {
    "AreaEggSlotsClient",
    "FirstAreaEgg_5262062450_5975783_Forest:Slot_003",
    "untitled.001",
}

local ChickenBossFarm = {
    enabled = false, stage = "idle",
    chickenName = "Chicken",
    altNames = {"galinha","hen","rooster","chicken"},
    waitTimeout = 120,
}

local function resolveChickenEggPath()
    local cur = Workspace
    for _, key in ipairs(CHICKEN_EGG_PATH) do
        if not cur then return nil end
        cur = cur:FindFirstChild(key)
    end
    return cur
end

local function getObjPos(obj)
    if not obj then return nil end
    if obj:IsA("BasePart") then return obj.Position end
    if obj:IsA("Model") or obj:IsA("Folder") then
        local prim = obj.PrimaryPart
        if prim then return prim.Position end
        local any = obj:FindFirstChildWhichIsA("BasePart", true)
        if any then return any.Position end
    end
    if obj:IsA("Attachment") then return obj.WorldPosition end
    local ok, p = pcall(function() return obj.Position end)
    if ok and typeof(p) == "Vector3" then return p end
    return nil
end

local function findChicken()
    local pathObj = resolveChickenEggPath()
    if pathObj then
        local pos = getObjPos(pathObj)
        if pos then
            return { obj = pathObj, pos = pos, isPath = true }
        end
    end
    local hints = {}
    for _, n in ipairs(ChickenBossFarm.altNames) do table.insert(hints, n:lower()) end
    table.insert(hints, ChickenBossFarm.chickenName:lower())
    local candidates = {}
    local function scan(container, depth)
        if depth > 6 or not container then return end
        for _, obj in ipairs(container:GetChildren()) do
            if obj:IsA("Model") or obj:IsA("BasePart") then
                local n = obj.Name:lower()
                for _, hint in ipairs(hints) do
                    if n:find(hint, 1, true) then
                        local pos = getObjPos(obj)
                        if pos then table.insert(candidates, {obj=obj, pos=pos}) end
                        break
                    end
                end
            end
            if obj:IsA("Model") or obj:IsA("Folder") then scan(obj, depth + 1) end
        end
    end
    scan(workspace, 0)
    local _, hrp = getChar()
    if hrp and #candidates > 1 then table.sort(candidates, function(a,b) return (a.pos-hrp.Position).Magnitude < (b.pos-hrp.Position).Magnitude end) end
    return candidates[1]
end

local function grabChickenEgg(chickenObj)
    if not chickenObj then return false end
    local fired = false
    for _, d in ipairs(chickenObj:GetDescendants()) do
        if d:IsA("ProximityPrompt") and d.Enabled then
            local nm = d.Name:lower()
            if nm:find("egg", 1, true) or nm:find("carry", 1, true) or nm:find("pickup", 1, true) or nm:find("take", 1, true) or nm:find("grab", 1, true) then
                pcall(function() d.HoldDuration = 0 fireproximityprompt(d) end)
                fired = true
            end
        end
    end
    if not fired then
        for _, d in ipairs(chickenObj:GetDescendants()) do
            if d:IsA("ProximityPrompt") and d.Enabled then
                pcall(function() d.HoldDuration = 0 fireproximityprompt(d) end)
                fired = true
            end
        end
    end
    local _, hrp = getChar()
    if hrp then
        local anchor = chickenObj:IsA("BasePart") and chickenObj or chickenObj:FindFirstChildWhichIsA("BasePart")
        if anchor then pcall(function() firetouchinterest(hrp, anchor, 0) firetouchinterest(hrp, anchor, 1) end) end
    end
    return fired
end

local function RunChickenSequence()
    if not ChickenBossFarm.enabled or ChickenBossFarm.stage == "done" then return end
    if HUB.dead then return end
    if ChickenBossFarm.stage == "idle" then
        ChickenBossFarm.stage = "going_chicken"
        if setStatus then setStatus("🐔 Indo no ovo da galinha...", true) end
        local chicken = findChicken()
        if not chicken then
            if setStatus then setStatus("❌ Ovo da galinha não achado (pulando)", false) end
            ChickenBossFarm.stage = "done"
            ChickenBossFarm.enabled = false
            return
        end
        flyTo(chicken.pos, 15)
        if setStatus then setStatus("🥚 Pegando ovo da galinha...", true) end
        local grabbed = false
        for attempt = 1, 20 do
            if not ChickenBossFarm.enabled then return end
            local c2 = findChicken()
            if c2 then
                grabChickenEgg(c2.obj)
                task.wait(0.12)
                if hasToolInHand() then grabbed = true break end
            else
                task.wait(0.2)
            end
        end
        ChickenBossFarm.stage = "waiting_hit"
        ChickenBossFarm.stageStartTime = tick()
        if setStatus then
            if grabbed then
                setStatus("⏳ Ovo na mão! Esperando galinha bater...", true)
            else
                setStatus("⏳ Esperando hit da galinha...", true)
            end
        end
        return
    end
    if ChickenBossFarm.stage == "waiting_hit" then
        if tick() - ChickenBossFarm.stageStartTime > ChickenBossFarm.waitTimeout then
            if setStatus then setStatus("⏱️ Timeout - iniciando farm", false) end
            ChickenBossFarm.stage = "done"
            ChickenBossFarm.enabled = false
        end
        return
    end
end

local KNOCKBACK_VELOCITY = 150
local KNOCKBACK_DISTANCE = 25
local KNOCKBACK_TIMEOUT  = 8

local _kbLastPos   = nil
local _kbLastCheck = 0
local _kbReadyAt   = 0

RunService.Heartbeat:Connect(function()
    if not ChickenBossFarm.enabled then return end
    if ChickenBossFarm.stage ~= "waiting_hit" then
        _kbLastPos = nil
        return
    end
    local _, hrp = getChar()
    if not hrp then return end

    local now = tick()

    if not _kbLastPos then
        _kbLastPos = hrp.Position
        _kbLastCheck = now
        _kbReadyAt = now + 1.0
        return
    end

    if now < _kbReadyAt then
        _kbLastPos = hrp.Position
        _kbLastCheck = now
        return
    end

    local vel = hrp.AssemblyLinearVelocity
    local horizVel = Vector3.new(vel.X, 0, vel.Z).Magnitude
    local moved = (hrp.Position - _kbLastPos).Magnitude

    local isKnockback = horizVel > KNOCKBACK_VELOCITY
        or moved > KNOCKBACK_DISTANCE
        or math.abs(vel.Y) > KNOCKBACK_VELOCITY

    if isKnockback then
        ChickenBossFarm.stage = "hit_processing"
        ChickenBossFarm.enabled = false
        _kbLastPos = nil
        if setStatus then setStatus("💥 Foi jogado! Teleportando...", true) end
        task.spawn(function()
            task.wait(0.35)
            if doHitBoost then pcall(doHitBoost) end
            ChickenBossFarm.stage = "done"
        end)
        return
    end

    if now - _kbLastCheck > KNOCKBACK_TIMEOUT then
        ChickenBossFarm.stage = "hit_processing"
        ChickenBossFarm.enabled = false
        _kbLastPos = nil
        if setStatus then setStatus("⏱️ Timeout — teleportando...", true) end
        task.spawn(function()
            if doHitBoost then pcall(doHitBoost) end
            ChickenBossFarm.stage = "done"
        end)
        return
    end

    _kbLastPos = hrp.Position
    _kbLastCheck = now
end)
